#include <iostream>
#include <vector>
using namespace std;

class MaxHeap {
    vector<int> heap;

    // Helper: maintain heap property after insertion
    void heapifyUp(int index) {
        while (index > 0) {
            int parent = (index - 1) / 2;
            if (heap[index] > heap[parent]) {
                swap(heap[index], heap[parent]);
                index = parent;
            } else break;
        }
    }

    // Helper: maintain heap property after deletion
    void heapifyDown(int index) {
        int size = heap.size();
        while (index < size) {
            int left = 2 * index + 1;
            int right = 2 * index + 2;
            int largest = index;

            if (left < size && heap[left] > heap[largest])
                largest = left;
            if (right < size && heap[right] > heap[largest])
                largest = right;

            if (largest != index) {
                swap(heap[index], heap[largest]);
                index = largest;
            } else break;
        }
    }

public:
    // Insert element
    void insert(int val) {
        heap.push_back(val);
        heapifyUp(heap.size() - 1);
    }

    // Delete root (maximum element)
    void deleteMax() {
        if (heap.empty()) {
            cout << "Heap is empty!\n";
            return;
        }
        cout << "Deleting max element: " << heap[0] << endl;
        heap[0] = heap.back();
        heap.pop_back();
        heapifyDown(0);
    }

    // Display heap elements
    void display() {
        cout << "Heap elements: ";
        for (int val : heap) cout << val << " ";
        cout << endl;
    }
};

int main() {
    MaxHeap h;
    h.insert(10);
    h.insert(40);
    h.insert(30);
    h.insert(60);
    h.insert(90);
    h.insert(50);

    h.display();

    h.deleteMax();
    h.display();

    h.deleteMax();
    h.display();

    return 0;
}
